 /*------------------------------------------------------------------*/
 /*                        Macro definitions                         */
 /*------------------------------------------------------------------*/

 #define ITIMER_REAL     1          /* Real time inteval timer       */
 #define ITIMER_VIRTUAL  2          /* Virtual inteval timer         */
 #define ITIMER_PROF     3          /* Profiling interval timer      */

 /*------------------------------------------------------------------*/
 /*                      Structure definitions                       */
 /*------------------------------------------------------------------*/

  struct timeval {
     long  tv_sec;                  /* second                        */
     long  tv_usec;                 /* microseconds                  */
   };

  struct itimerval {
     struct timeval it_interval;    /* timer interval                */
     struct timeval it_value;       /* current value                 */
   };

  struct timezone {
      int  tz_minuteswest;          /* minutes west of Greenwich     */
      int  tz_dsttime;              /* daylight savings time flag    */
   };

 /*------------------------------------------------------------------*/
 /*                       Function prototypes                        */
 /*------------------------------------------------------------------*/

  QBFC_EXTERN int select( int, fd_set *, fd_set *, fd_set *,
                          struct timeval * );

  QBFC_EXTERN int adjtime( struct timeval *,
                           struct timeval * );

  QBFC_EXTERN int gettimeofday( struct timeval *,
                                struct timezone * );

  QBFC_EXTERN int settimeofday( struct timeval *,
                                struct timezone * );

  QBFC_EXTERN int getitimer( int, struct itimerval * );

  QBFC_EXTERN int setitimer( int, const struct itimerval *,
                             struct itimerval * );

  QBFC_EXTERN int Qp0zAdjTime( struct timeval *,
                               struct timeval * );             /*@A6A*/

  QBFC_EXTERN int Qp0zGetTimeofDay( struct timeval *,
                                    struct timezone * );       /*@A6A*/

  QBFC_EXTERN int Qp0zSetTimeofDay( struct timeval *,
                                    struct timezone * );       /*@A6A*/
