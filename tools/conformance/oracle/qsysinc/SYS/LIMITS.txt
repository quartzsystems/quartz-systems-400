#ifndef __sys_limits_h
#define __sys_limits_h  1

/*********************************************
     Minimum Values
*********************************************/

#define _POSIX_LINK_MAX      8     /* The value of a file's link
                                    * count                        */
#define _POSIX_NAME_MAX     14     /* The number of bytes in a
                                    * filename                     */
#define _POSIX_NGROUPS_MAX   0     /* The number of simultaneous
                                    * supplementary group IDS per
                                    * process                      */
#define _POSIX_OPEN_MAX     16     /* The number of files that one
                                    * process can have open at one
                                    * time                         */
#define _POSIX_PATH_MAX    255     /* The number of bytes in a
                                    * pathname                     */
#define _POSIX_SSIZE_MAX 32767     /* The value that can be stored
                                    * in an object of type ssize_t */
#define _POSIX_STREAM_MAX    8     /* The number of streams that
                                    * one process can have open at
                                    * one time                     */

#define _POSIX2_COLL_WEIGHTS_MAX  3 /* The maximum number of weights
                                     * that can be assigned to an
                                     * entry of the LC_COLLATE order
                                     * keyword in the locale definition
                                     * file.
                                     *                         @A1A
                                     *                         @A2C */
#define COLL_WEIGHTS_MAX          _POSIX2_COLL_WEIGHTS_MAX  /* @A1A */

#define _POSIX_SEM_NSEMS_MAX 256                            /* @A8A */
#define _POSIX_SEM_VALUE_MAX 32767                          /* @A8A */

/*********************************************
   NAMED AND UNNAMED SEMAPHORE DEFINITIONS
**********************************************/
/* not POSIX.1 or POSIX.1a defined symbolic constants */
#if (!(defined _POSIX_SOURCE) && !(defined _POSIX1_SOURCE))
#define SEM_VALUE_MAX 2147483647                            /* @A8A */
#define SEM_NSEMS_MAX 2147483647                            /* @A8A */
#define SEM_NAME_MAX 1024                                   /* @A8A */
#endif

/*********************************************
     Message Catalog Values
*********************************************/
/* not POSIX.1 or POSIX.1a defined symbolic constants */
#if (!(defined _POSIX_SOURCE) && !(defined _POSIX1_SOURCE))
  #define NL_TEXTMAX     16384      /* The maximum length of a message
                                     * in a message catalog.    @A6A
                                                                @A9C*/
  #define NL_MSGMAX      65535      /* The maximum number of messages
                                     * per set in a message catalog
                                     *                          @A6A*/
  #define NL_SETMAX      65535      /* The maximum number of sets in
                                     * a message catalog.       @A6A*/
#endif

/*********************************************
     Runtime Increasable Values
*********************************************/

#define NGROUPS_MAX         16     /* Maximum number of simultaneous
                                    * supplementary group IDs per
                                    * process                      */


/*********************************************
     Invariant Values
*********************************************/

#define SSIZE_MAX      INT_MAX     /* Maximum value that can be
                                    * stored in an object of type
                                    * ssize_t                      */

/* POSIX.1a defined symbolic constant */
#if ((!(defined _POSIX_SOURCE) && !(defined _POSIX1_SOURCE)) \
     || ((defined _POSIX1_SOURCE) && (_POSIX1_SOURCE != 1)))
  #define POSIX_SYMLOOP       16     /* Number of symbolic links that
                                      * can be traversed in the
                                      * resolution of a pathname in
                                      * the absence of a loop.
                                      */
#endif /* POSIX.1a defined symbolic constant */

/*********************************************
    XPG4 Support constant
*********************************************/

#define NL_ARGMAX   256           /* Maximum number of positional
                                     parameters that can be used in
                                     the printf/scanf family of
                                     function                 @A7A*/

#endif /* __sys_limits_h */
