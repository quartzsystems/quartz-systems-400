#include <stdlib.h>                   /* wchar_t                     */
#include <qlg.h>                      /* Qlg_Path_Name               */
#if (__OS400_TGTVRM__>=510)  /* @B1A*/
#pragma datamodel(P128)      /* @B1A*/
#endif                       /* @B1A*/

/*********************************************************************/
/* Macros to define sunc_format constants                            */
/*********************************************************************/

#define SO_UNC_DEFAULT    0      /* sunc_qlg is ignored.             */
                                 /* Default UNICODE CCSID is used.   */
                                 /* sunc_path.wide contains the path.*/

#define SO_UNC_USE_QLG    1      /* sunc_qlg is used to define the   */
                                 /* CCSID and the format of sunc_path*/

/* Define a constant for the size of the fixed portion of the        */
/* address before the sunc_path union.                               */

#define SO_UNC_FIXED_SIZE (16 + sizeof(Qlg_Path_Name_T))

/*********************************************************************/
/* Structures                                                        */
/*********************************************************************/

struct sockaddr_unc {                 /* socket address (UNIX CCSID) */

   short           sunc_family;       /* address family AF_UNIX_CCSID*/
   short           sunc_format;       /* format of sunc_path         */
   char            sunc_zero[12];     /* reserved - must be 0x00's   */
   Qlg_Path_Name_T sunc_qlg;          /* path name national language */
   union {
      char            unix[126];
      wchar_t         wide[126];
      char*           p_unix;
      wchar_t*        p_wide;
   }               sunc_path;         /* path name                   */
};

#endif                       /* @B1A*/
