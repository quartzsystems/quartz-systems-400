#define SUN_LEN(qso_su)            \
  (sizeof(*(qso_su)) -             \
   sizeof((qso_su)->sun_path) +    \
   strlen((qso_su)->sun_path))        /* determine actual length
                                         of a AF_UNIX domain address
                                         in which the path name is
                                         NULL-terminated and in the
                                         single byte coded character
                                         set (SBCS)                  */

#if ((_XOPEN_SOURCE+1>520) && !defined(COMPAT_43))
/* _XOPEN_SOURCE is defined to be >= 520 & not BSD 4.3 compatibility */
#include <inttypes.h>

#ifndef _SA_FAMILY_T
  typedef unsigned char sa_family_t;
  #define _SA_FAMILY_T
#endif

struct sockaddr_un {         /* socket address (UNIX)       */

   uint8_t     sun_len;               /* address length              */
   sa_family_t sun_family;            /* address familiy (AF_UNIX)   */
   char        sun_path[126];         /* path name in job's CCSID    */
};

#else
/* _XOPEN_SOURCE is not defined to be >= 520 or BSD 4.3 compatibility*/

struct sockaddr_un {         /* socket address (UNIX)       */

   short     sun_family;              /* address familiy (AF_UNIX)   */
   char      sun_path[126];           /* path name in job's CCSID    */
};

#endif /* End changes for _XOPEN_SOURCE                          @A2A*/
