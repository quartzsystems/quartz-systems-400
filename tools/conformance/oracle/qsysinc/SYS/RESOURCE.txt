#pragma datamodel(P128)

/**********************************************************************/
/* Constants                                                          */
/**********************************************************************/

#define RLIM_NLIMITS    7    /* Number of limits */

#define RLIMIT_FSIZE    0    /* File size in bytes */
#define RLIMIT_NOFILE   1    /* Value of largest descriptor number */
#define RLIMIT_CORE     2    /* Core file size in bytes */
#define RLIMIT_CPU      3    /* CPU time in seconds */
#define RLIMIT_DATA     4    /* Data segment size in bytes */
#define RLIMIT_STACK    5    /* Stack size in bytes */
#define RLIMIT_AS       6    /* Total available memory in bytes */

#define RLIM_INFINITY   (0xFFFFFFFFFFFFFFFFLL) /* No limit */
#define RLIM_SAVED_MAX  (0xFFFFFFFFFFFFFFFELL) /* Set to current hard limit */
#define RLIM_SAVED_CUR  (0xFFFFFFFFFFFFFFFDLL) /* Set to current soft limit */

/**********************************************************************/
/* Types                                                              */
/**********************************************************************/

typedef unsigned long long int rlim_t;

/**********************************************************************/
/* Structures                                                         */
/**********************************************************************/

struct rlimit {
  rlim_t rlim_cur;           /* Current or soft limit */
  rlim_t rlim_max;           /* Maximum or hard limit */
};
