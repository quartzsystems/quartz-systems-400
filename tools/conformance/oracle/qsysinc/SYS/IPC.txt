#include <sys/types.h>
#include <qlg.h>             /* @A3A*/

#pragma datamodel(P128)      /* @B1A*/

/*********************************************************************/
/* Constants                                                         */
/*********************************************************************/

/* Mode bits */
#define IPC_CREAT     0X200  /* Create entry if key does not exist */
#define IPC_EXCL      0X400  /* Fail if key exists */
#define IPC_NOWAIT    0X800  /* Error if request must wait */

/* Keys */
#define IPC_PRIVATE (key_t)0 /* Private key */

/* Command definitions for msgctl(), semctl(), shmctl() */
#define IPC_RMID        0    /* Remove identifier */
#define IPC_SET         1    /* Set options */
#define IPC_STAT        2    /* Get options */

/*********************************************************************/
/* Structures                                                        */
/*********************************************************************/

typedef struct ipc_perm {
  uid_t  uid;                /* Owner's user ID */
  gid_t  gid;                /* Owner's group ID */
  uid_t  cuid;               /* Creator's user ID */
  gid_t  cgid;               /* Creator's group ID */
  mode_t mode;               /* Permissions */
} ipc_perm_t;
